struct Point
{
    i32 x
    i32 y

    fn d2 := () -> i32
    {
        i32 result{x * x + y * y}
        exit result
    }
}

struct Container
{
    Point point

    fn getPoint := () -> Point
    {
        exit point
    }
}

Point p{3, 4}
Container c{p}
i32 x{c.getPoint().x}
i32 distance{c.getPoint().d2()}
exit distance
