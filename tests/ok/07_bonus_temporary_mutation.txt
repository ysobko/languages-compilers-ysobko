struct Point
{
    i32 mut x
    i32 mut y

    fn shift := (i32 dx) -> i32
    {
        x := x + dx
        exit x
    }
}

struct Container
{
    Point point

    fn getPoint := () -> Point
    {
        exit point
    }
}

Point p{3, 4}
Container c{p}

i32 changed{c.getPoint().shift(10)}
i32 original{c.point.x}

print(changed)
print(original)

exit original
